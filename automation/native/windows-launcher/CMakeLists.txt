cmake_minimum_required(VERSION 3.25)
project(KiCadWindowsLaunchers LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "Windows launcher builds require Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(kicad-automation-launcher WIN32 launcher.cpp)
target_compile_definitions(kicad-automation-launcher PRIVATE KICAD_LAUNCH_GUI)
add_executable(kicad-automation-mcp-launcher launcher.cpp)
foreach(target kicad-automation-launcher kicad-automation-mcp-launcher)
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../../thirdparty/nlohmann_json")
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE)
    target_link_libraries(${target} PRIVATE bcrypt shell32 user32)
    install(TARGETS ${target} RUNTIME DESTINATION bin)
endforeach()
option(KICAD_BUILD_LAUNCHER_FIXTURE "Build the non-installable QA argument/stdio probe" OFF)
if(KICAD_BUILD_LAUNCHER_FIXTURE)
    add_executable(launcher-probe "${CMAKE_CURRENT_LIST_DIR}/../../tests/fixtures/windows-launcher-probe.cpp")
    target_include_directories(launcher-probe PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../../thirdparty/nlohmann_json")
endif()
